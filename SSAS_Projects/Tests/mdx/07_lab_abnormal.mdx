-- Лабораторія: частка аналізів з відхиленням за аналізом і віковою групою
SELECT NON EMPTY [Вікова група].[Вікова група].[Вікова група].MEMBERS * { [Measures].[Кількість аналізів], [Measures].[Частка аналізів з відхиленням] } ON COLUMNS,
       NON EMPTY [Аналіз].[Аналіз].[Аналіз].MEMBERS ON ROWS
FROM [Поліклініка]
