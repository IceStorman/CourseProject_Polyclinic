-- Топ-10 лікарів за виручкою у 2025 році з рейтингом
SELECT { [Measures].[Виручка], [Measures].[Рейтинг лікаря за виручкою], [Measures].[Проведені візити],
         [Measures].[Середня вартість візиту], [Measures].[Частка виручки] } ON COLUMNS,
       [Топ-10 лікарів за виручкою] ON ROWS
FROM [Поліклініка]
WHERE [Дата].[Рік].&[2025]
